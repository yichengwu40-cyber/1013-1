ESP32 Arduino 完整控制程式碼

相容於最新 ESP32 Arduino Core v3.x 韌體，具備序列埠即時回報功能。

const int LDR_PIN = 39;     // 光感測器輸入 (ADC1_CH3)
const int POT_PIN = 34;     // 可變電阻輸入 (ADC1_CH6)
const int LED_PIN = 12;     // LED 輸出 (PWM)

// 設定 LEDC PWM 參數 (ESP32 Arduino Core v3.x 寫法)
const int freq = 5000;      // 頻率 5 kHz
const int resolution = 8;   // 解析度 8-bit (0-255)
const int ledChannel = 0;   // 通道

void setup() {
  Serial.begin(115200);

  // 初始化 PWM 設定
  ledcAttachChannel(LED_PIN, freq, resolution, ledChannel);
}

void loop() {
  // 1. 讀取感測器數值 (0 ~ 4095)
  int ldrValue = analogRead(LDR_PIN);
  int potValue = analogRead(POT_PIN);

  // 2. 映射邏輯：環境越暗 (LDR 越小)，LED 亮度越大 (PWM 越大)
  int ledBrightness = map(ldrValue, 0, 4095, 255, 0);
  
  // 限制數值在 0 到 255 之間
  ledBrightness = constrain(ledBrightness, 0, 255);

  // 3. 輸出 PWM 控制 LED 亮度
  ledcWrite(LED_PIN, ledBrightness);

  // 4. JSON 格式輸出至序列埠（方便網頁 WebSerial 讀取）
  Serial.print("{\"ldr\":");
  Serial.print(ldrValue);
  Serial.print(",\"pot\":");
  Serial.print(potValue);
  Serial.print(",\"led\":");
  Serial.print(ledBrightness);
  Serial.println("}");

  delay(100);
}
